# Supervisor accepts exactly one column-zero `profile ` declaration as the
# primary App profile. Keep every secondary declaration below indented by two
# spaces; AppArmor still parses them as independent global profiles, not local
# child profiles, so the existing directed Px boundaries and names stay intact.
# The public launch profiles are intentionally short-lived bootstraps.  They
# validate the image-owned shell wrapper, clear the inherited environment with
# the image-owned env binary, and make the one permitted transition into the
# native CLI runtime.  Keeping the bootstrap separate means model-controlled
# tools cannot re-exec antigravity-real in the OAuth-readable runtime profile.

  capability setuid,

  /usr/local/libexec/antigravity-interactive-restricted r,

  /usr/local/libexec/antigravity-real Px -> antigravity_home_assistant-interactive-runtime-restricted,
  /lib/** rm,
  /lib64/** rm,
  /etc/ld.so.cache r,

  /dev/null rw,
  /dev/tty rw,
  /proc/ r,
  /proc/** r,
}
